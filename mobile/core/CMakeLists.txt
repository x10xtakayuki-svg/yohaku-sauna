cmake_minimum_required(VERSION 3.22)
project(YohakuAudio LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(yohaku_core STATIC SaunaEngine.cpp)
target_include_directories(yohaku_core PUBLIC .)
target_compile_options(yohaku_core PRIVATE -Wall -Wextra -Wpedantic)
if(NOT ANDROID)
  add_executable(core_tests tests.cpp)
  target_link_libraries(core_tests PRIVATE yohaku_core)
  # Assertions are intentionally active in optimized audio tests.
  target_compile_options(core_tests PRIVATE -UNDEBUG)
  enable_testing()
  add_test(NAME background_audio COMMAND core_tests)
endif()
